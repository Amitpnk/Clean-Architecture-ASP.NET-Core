# EF Core migrations (.NET 10)
# dotnet-ef is a local tool pinned in dotnet-tools.json - restore it once per clone:
dotnet tool restore

# All tables (Identity + application) live in ApplicationDbContext; migrations go to CleanArch.Persistence/Migrations.
dotnet ef migrations add <MigrationName> --project src/CleanArch.Persistence --startup-project src/CleanArch.Api --context ApplicationDbContext --output-dir Migrations

# Apply migrations (the API also applies pending migrations on startup).
dotnet ef database update --project src/CleanArch.Persistence --startup-project src/CleanArch.Api --context ApplicationDbContext

# CI check: fails if the model changed without a migration.
dotnet ef migrations has-pending-model-changes --project src/CleanArch.Persistence --startup-project src/CleanArch.Api --context ApplicationDbContext
